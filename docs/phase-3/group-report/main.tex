  \begin{sloppypar}\noindent\textbf{Mã artefact:} \texttt{#2}. #3\end{sloppypar}\medskip

\newcommand{\SubsystemClassDesign}[2]{%
  \subsection{#1 - #2}
  \ArtifactBrief{CLASS DIAGRAM}{CD-#1}{Bổ sung lớp biên, lớp ứng dụng, lớp miền và lớp hạ tầng cần thiết; thể hiện quan hệ, multiplicity, visibility và kiểu dữ liệu.}{Class diagram #1 - #2}
  \noindent Class diagram này là artefact chuẩn của phân hệ. Các use case bên dưới chỉ trích các class/method thực sự tham gia, không sao chép lại toàn bộ diagram.\par
}

\newcommand{\UseCaseDesign}[5]{%
  \subsubsection{#1 -- #2}
  \paragraph{Nguồn thiết kế và truy vết}
  \begin{sloppypar}Màn hình #3; behavioral diagram \texttt{SEQ-#1}, \texttt{ACT-#1}; requirement và business rule liên quan: #4.\end{sloppypar}
  \paragraph{Lớp và thành phần tham gia}
  Liệt kê boundary/controller, application service, domain object/service, repository và external adapter thực sự tham gia \texttt{#1}; nêu trách nhiệm và dependency direction của từng thành phần.
  \paragraph{Đặc tả phương thức và giao diện}
  Với mỗi method/API/event, ghi signature, input/output, precondition, postcondition, exception/error, side effect, transaction boundary và mã UC/FR/BR liên quan. Không liệt kê method không xuất hiện trong luồng thiết kế.
  \paragraph{Kiểm thử liên quan}
  Ưu tiên kiểm chứng #5. Mỗi test case ghi dữ liệu đầu vào, bước thực hiện, expected result, actual result và evidence khi đã chạy.
}
